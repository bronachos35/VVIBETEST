-- Services
local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")
local LocalPlayer = Players.LocalPlayer
local PlayerGui = LocalPlayer:WaitForChild("PlayerGui")

-- Script Link Configuration
local loadstringText = 'loadstring(game:HttpGet("https://raw.githubusercontent.com/bronachos35/Vibe-ftap-obf/refs/heads/main/Script"))()'

-- Clean up previous UI instance
if PlayerGui:FindFirstChild("OutdatedNoticeUI") then
    PlayerGui.OutdatedNoticeUI:Destroy()
end

-- ScreenGui Setup
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "OutdatedNoticeUI"
ScreenGui.ResetOnSpawn = false
ScreenGui.Parent = PlayerGui

-- Main Container (Centered Overlay)
local Frame = Instance.new("Frame")
Frame.Size = UDim2.new(0, 380, 0, 220)
Frame.Position = UDim2.new(0.5, -190, 0.5, -110)
Frame.BackgroundColor3 = Color3.fromRGB(18, 18, 22)
Frame.BorderSizePixel = 0
Frame.ClipsDescendants = true
Frame.Parent = ScreenGui

local UICorner = Instance.new("UICorner")
UICorner.CornerRadius = UDim.new(0, 14)
UICorner.Parent = Frame

local UIStroke = Instance.new("UIStroke")
UIStroke.Thickness = 1.5
UIStroke.Color = Color3.fromRGB(45, 45, 55)
UIStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
UIStroke.Parent = Frame

-- Subtle Header Glow Line
local AccentLine = Instance.new("Frame")
AccentLine.Size = UDim2.new(1, 0, 0, 3)
AccentLine.Position = UDim2.new(0, 0, 0, 0)
AccentLine.BackgroundColor3 = Color3.fromRGB(255, 70, 70)
AccentLine.BorderSizePixel = 0
AccentLine.Parent = Frame

-- Warning Badge Icon Container
local IconFrame = Instance.new("Frame")
IconFrame.Size = UDim2.new(0, 42, 0, 42)
IconFrame.Position = UDim2.new(0.5, -21, 0, 16)
IconFrame.BackgroundColor3 = Color3.fromRGB(35, 25, 25)
IconFrame.BorderSizePixel = 0
IconFrame.Parent = Frame

local IconCorner = Instance.new("UICorner")
IconCorner.CornerRadius = UDim.new(1, 0)
IconCorner.Parent = IconFrame

local IconLabel = Instance.new("TextLabel")
IconLabel.Size = UDim2.new(1, 0, 1, 0)
IconLabel.BackgroundTransparency = 1
IconLabel.Text = "⚠️"
IconLabel.TextSize = 20
IconLabel.Parent = IconFrame

-- Main Status Text
local TitleLabel = Instance.new("TextLabel")
TitleLabel.Size = UDim2.new(1, -40, 0, 50)
TitleLabel.Position = UDim2.new(0, 20, 0, 64)
TitleLabel.BackgroundTransparency = 1
TitleLabel.Text = "SCRIPT IS OUTDATED USE NEW ONE"
TitleLabel.TextColor3 = Color3.fromRGB(240, 240, 245)
TitleLabel.TextSize = 15
TitleLabel.Font = Enum.Font.GothamBold
TitleLabel.TextWrapped = true
TitleLabel.Parent = Frame

-- Button Container
local ButtonContainer = Instance.new("Frame")
ButtonContainer.Size = UDim2.new(1, -40, 0, 80)
ButtonContainer.Position = UDim2.new(0, 20, 1, -92)
ButtonContainer.BackgroundTransparency = 1
ButtonContainer.Parent = Frame

local UIListLayout = Instance.new("UIListLayout")
UIListLayout.SortOrder = Enum.SortOrder.LayoutOrder
UIListLayout.Padding = UDim.new(0, 8)
UIListLayout.Parent = ButtonContainer

-- Copy Button
local CopyButton = Instance.new("TextButton")
CopyButton.LayoutOrder = 1
CopyButton.Size = UDim2.new(1, 0, 0, 38)
CopyButton.BackgroundColor3 = Color3.fromRGB(255, 75, 75)
CopyButton.Text = "Copy New Script"
CopyButton.TextColor3 = Color3.fromRGB(255, 255, 255)
CopyButton.TextSize = 14
CopyButton.Font = Enum.Font.GothamMedium
CopyButton.AutoButtonColor = false
CopyButton.Parent = ButtonContainer

local CopyCorner = Instance.new("UICorner")
CopyCorner.CornerRadius = UDim.new(0, 8)
CopyCorner.Parent = CopyButton

-- Remove GUI Button (Initially Hidden)
local RemoveButton = Instance.new("TextButton")
RemoveButton.LayoutOrder = 2
RemoveButton.Size = UDim2.new(1, 0, 0, 34)
RemoveButton.BackgroundColor3 = Color3.fromRGB(30, 30, 38)
RemoveButton.Text = "Remove GUI"
RemoveButton.TextColor3 = Color3.fromRGB(170, 170, 180)
RemoveButton.TextSize = 13
RemoveButton.Font = Enum.Font.Gotham
RemoveButton.Visible = false
RemoveButton.AutoButtonColor = false
RemoveButton.Parent = ButtonContainer

local RemoveCorner = Instance.new("UICorner")
RemoveCorner.CornerRadius = UDim.new(0, 8)
RemoveCorner.Parent = RemoveButton

local RemoveStroke = Instance.new("UIStroke")
RemoveStroke.Thickness = 1
RemoveStroke.Color = Color3.fromRGB(50, 50, 60)
RemoveStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
RemoveStroke.Parent = RemoveButton

-- Button Click Actions & Logic
CopyButton.MouseButton1Click:Connect(function()
    -- Copy loadstring to clipboard
    pcall(function()
        setclipboard(loadstringText)
    end)
    
    -- Update Header Accent & Icon
    AccentLine.BackgroundColor3 = Color3.fromRGB(46, 204, 113)
    IconFrame.BackgroundColor3 = Color3.fromRGB(20, 40, 28)
    IconLabel.Text = "✓"
    
    -- Update Notice Text
    TitleLabel.Text = "SCRIPT IS OUTDATED USE NEW ONE"
    TitleLabel.TextColor3 = Color3.fromRGB(46, 204, 113)
    
    -- Update Copy Button state
    CopyButton.Text = "Copied to Clipboard!"
    CopyButton.BackgroundColor3 = Color3.fromRGB(35, 155, 86)
    
    -- Reveal Remove GUI Button
    RemoveButton.Visible = true
end)

-- Remove GUI Click Event
RemoveButton.MouseButton1Click:Connect(function()
    -- Smooth exit animation
    local tweenInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
    local tween = TweenService:Create(Frame, tweenInfo, {
        Size = UDim2.new(0, 380, 0, 0),
        Position = UDim2.new(0.5, -190, 0.5, 0)
    })
    
    tween:Play()
    tween.Completed:Connect(function()
        ScreenGui:Destroy()
    end)
end)
